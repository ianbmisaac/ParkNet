const COLORS = { red:"#d33a2c", black:"#20242c", orange:"#e8751a", gold:"#d4a017", green:"#3f9e4f", blue:"#2e7ad6" };
// el color es el elemento string del arreglo de la conexion ("red","black",...)
const colorName = c => {
  if (!Array.isArray(c)) return "#888";
  const s = c.find(x => typeof x === "string" && COLORS[x]);
  return s ? COLORS[s] : "#888";
};

const BB = { x: 24, y: 148, w: 1132, h: 344, cols: 60 };
const colX = c => BB.x + 34 + (c - 1) * ((BB.w - 68) / (BB.cols - 1));
const railY = r => r === "tp" ? BB.y + 10 : BB.y + 42;
const rowsA = [0,1,2,3,4].map(i => BB.y + 84 + i * 31);   // 232..356
const rowsB = [0,1,2,3,4].map(i => BB.y + 244 + i * 22);   // 392..480
const rowIndex = { a:0,b:1,c:2,d:3,e:4, f:0,g:1,h:2,i:3,j:4 };

function pinAnchor(pin) {
  const map = { "11": 0, "12": 1, "13": 2, "A0": 3, "A4": 4, "A5": 5, "GND.1": 6, "GND": 6 };
  const i = map[pin] ?? 6;
  return { x: UNO.x + UNO.w, y: UNO.y + 34 + i * 26, label: pin };
}
function gridHole(col, row) {
  if (row === "tp") return { x: colX(col), y: railY("tp"), label: "TP" + col };
  if (row === "tn") return { x: colX(col), y: railY("tn"), label: "TN" + col };
  const r = rowIndex[row];
  const arr = "abcde".includes(row) ? rowsA : rowsB;
  return { x: colX(col), y: arr[r], label: col + row };
}
function partPin(key) {
  const [id, pin] = key.split(":");
  const r = PARTS[id], o = PIN_POS[key];
  if (!r || !o) return { x: 0, y: 0, label: key };
  let x, y;
  if (o.s === 't')      { x = r.x + r.w * o.dx; y = r.y; }
  else if (o.s === 'b') { x = r.x + r.w * o.dx; y = r.y + r.h; }
  else if (o.s === 'r') { x = r.x + r.w;       y = r.y + r.h * o.dy; }
  else                  { x = r.x;             y = r.y + r.h * o.dy; }
  return { x, y, label: key };
}
function parseEnd(e) {
  const id = e.replace(/^(uno|bb):/, "");
  if (e.startsWith("uno:")) return pinAnchor(id);
  const rail = id.match(/^(tp|tn)\.(\d+)$/);
  if (rail) return gridHole(+rail[2], rail[1]);
  const cell = id.match(/^(\d+)t\.([a-j])$/);
  if (cell) return gridHole(+cell[1], cell[2]);
  const cell2 = id.match(/^(\d+)\.([a-j])$/);
  if (cell2) return gridHole(+cell2[1], cell2[2]);
  return partPin(e);
}
